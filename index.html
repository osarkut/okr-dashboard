const SHEET_ENDPOINT = "https://docs.google.com/spreadsheets/d/184A7LoWQ637eHfqM3TLeXZ53BSTngdnUEUyS-U9l31c/gviz/tq?tqx=out:json&sheet=32_ai"; async function syncDataFromAppsScript() { const statusEl = document.getElementById('sync-status'); if (statusEl) { statusEl.innerText = "Syncing Sheet Data..."; statusEl.className = "text-xs text-yellow-400 font-semibold bg-yellow-950/60 border border-yellow-800 px-3.5 py-1.5 rounded-full"; } try { const response = await fetch(SHEET_ENDPOINT); const rawText = await response.text(); // 1. Extract JSON payload from Google's response wrapper const jsonMatch = rawText.match(/google\.visualization\.Query\.setResponse\(([\s\S]*)\);/); if (!jsonMatch || !jsonMatch[1]) { throw new Error("Invalid response structure from Google Sheets"); } const parsedData = JSON.parse(jsonMatch[1]); const table = parsedData.table; // 2. Map Google columns to key names const headers = table.cols.map(col => col.label); // 3. Format rows into clean objects const rows = table.rows.map(row => { const rowObj = {}; row.c.forEach((cell, idx) => { const key = headers[idx] || `col_${idx}`; if (cell) { rowObj[key] = cell.f !== undefined ? cell.f : cell.v; } else { rowObj[key] = null; } }); return rowObj; }); console.log("Parsed rows successfully:", rows); // 4. Update UI Dashboard Counters const validTasks = rows.filter(r => r["Task"]); const greenTasksCount = rows.filter(r => r["Status"] === "Completed" || r["Status"] === "Done").length; const greenTasksEl = document.getElementById('green-tasks'); if (greenTasksEl) greenTasksEl.innerText = `${validTasks.length} Tasks`; const overallScoreEl = document.getElementById('overall-score'); if (overallScoreEl) { const percentage = validTasks.length > 0 ? Math.round((greenTasksCount / validTasks.length) * 100) : 0; overallScoreEl.innerText = `${percentage}%`; } if (statusEl) { statusEl.innerText = "● Tab 32_ai Live Sync Active"; statusEl.className = "text-xs text-green-400 font-semibold bg-green-950/60 border border-green-800 px-3.5 py-1.5 rounded-full"; } } catch (err) { console.error("Dashboard Sync Failed:", err); if (statusEl) { statusEl.innerText = "● Sync Error"; statusEl.className = "text-xs text-red-400 font-semibold bg-red-950/60 border border-red-800 px-3.5 py-1.5 rounded-full"; } } } // Ensure function executes on page load document.addEventListener("DOMContentLoaded", syncDataFromAppsScript);
