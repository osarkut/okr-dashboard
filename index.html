Strip out the google.visualization.Query.setResponse wrapper const jsonMatch = rawText.match(/google\.visualization\.Query\.setResponse\(([\s\S]*)\);/); if (!jsonMatch || !jsonMatch[1]) { throw new Error("Could not parse Google Sheets response"); } const parsedData = JSON.parse(jsonMatch[1]); const table = parsedData.table; // 2. Extract column labels const headers = table.cols.map(col => col.label); // 3. Map rows to JavaScript Objects const rows = table.rows.map(row => { const rowData = {}; row.c.forEach((cell, index) => { const header = headers[index]; if (header) { // Use formatted text 'f' if available, otherwise raw value 'v' rowData[header] = cell ? (cell.f !== undefined ? cell.f : cell.v) : null; } }); return rowData; }); console.log("Successfully parsed rows from Tab 32_ai:", rows); // 4. Update UI Status const totalTasks = rows.filter(r => r["Task"]).length; document.getElementById('green-tasks').innerText = `${totalTasks} Sheet Rows`; document.getElementById('overall-score').innerText = "100%"; statusEl.innerText = "● Tab 32_ai Direct Sync Active"; statusEl.className = "text-xs text-green-400 font-semibold bg-green-950/60 border border-green-800 px-3.5 py-1.5 rounded-full"; } catch (err) { console.error("Direct sheet sync error:", err); statusEl.innerText = "● Sheet Sync Failed"; statusEl.className = "text-xs text-red-400 font-semibold bg-red-950/60 border border-red-800 px-3.5 py-1.5 rounded-full"; } }
